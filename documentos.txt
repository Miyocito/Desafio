def get_embedding(text, model_id=bedrock_model_embeddings):
    """Invoca Bedrock para generar embedding y devuelve la lista de floats."""
    payload = {"inputText": text}
    resp = bedrock.invoke_model(modelId=model_id, body=json.dumps(payload))
    body_bytes = resp["body"].read()
    result = json.loads(body_bytes.decode("utf-8"))
    # Nota: la estructura exacta puede variar por modelo; lo más común:
    # result["embedding"] -> list[float]
    # Si no existe, imprime result para inspección.
    emb = result.get("embedding") or result.get("embeddings")
    if emb is None:
        # para debug: imprime la respuesta del modelo
        raise RuntimeError(f"No se encontró 'embedding' en la respuesta de Bedrock: {result}")
    return emb

def ask_bedrock(question, context, model_id=bedrock_model_llm):
    """Pregunta al LLM pasando contexto (RAG)."""
    prompt = f"Usa el siguiente contexto:\n{context}\n\nPregunta: {question}\nRespuesta:"
    payload = {"inputText": prompt}
    resp = bedrock.invoke_model(modelId=model_id, body=json.dumps(payload))
    body_bytes = resp["body"].read()
    result = json.loads(body_bytes.decode("utf-8"))
    # la clave para texto suele ser "outputText" o similar según modelo -> revisa la doc del modelo
    return result.get("outputText") or result.get("content") or json.dumps(result)

def search_context(question, k=3):
    q_emb = get_embedding(question)
    
    response = client.collections.get(class_name).query.near_vector(
        near_vector=q_emb,
        limit=k
    )
    
    context = "\n".join([o.properties["content"] for o in response.objects])
    return context
